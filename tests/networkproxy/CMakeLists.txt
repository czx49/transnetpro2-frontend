cmake_minimum_required(VERSION 3.16)
project(BackendProxyTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 REQUIRED COMPONENTS Core Network Test)
enable_testing()
add_executable(backend_proxy_tests backend_proxy_tests.cpp)
target_include_directories(backend_proxy_tests PRIVATE ../../src/httpapis)
target_link_libraries(backend_proxy_tests PRIVATE Qt5::Core Qt5::Network Qt5::Test)
add_test(NAME BackendProxyRouting COMMAND backend_proxy_tests)

if(QGIS_DIR)
    find_package(Qt5 REQUIRED COMPONENTS Widgets Xml Sql Concurrent)
    add_executable(qgis_proxy_tests qgis_proxy_tests.cpp)
    target_include_directories(qgis_proxy_tests PRIVATE ../../src/httpapis "${QGIS_DIR}/include")
    target_link_libraries(qgis_proxy_tests PRIVATE Qt5::Core Qt5::Network Qt5::Test
        Qt5::Widgets Qt5::Xml Qt5::Sql Qt5::Concurrent "${QGIS_DIR}/lib/qgis_core.lib")
    if(MSVC)
        target_compile_definitions(qgis_proxy_tests PRIVATE _USE_MATH_DEFINES NOMINMAX)
        target_compile_options(qgis_proxy_tests PRIVATE /utf8)
    endif()
    add_test(NAME QgisBackendProxyRouting COMMAND qgis_proxy_tests)
    set_tests_properties(QgisBackendProxyRouting PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
